; DEMASTER: the time of day and colour overrides in battles and on the world map (SCENE_LIGHT in demaster.conf).
; Read again within a second of saving it. Colours are red green blue in linear light: 1 1 1 changes nothing.
;
; night       = auto | 0 | 1   does the night come here. auto: the world map always; a battle when it started on
;                              the world map or in a field open to the sky (sky = 1 in its _pbr.txt). Battles from
;                              other fields stay as painted; from a field with a fixed time (time = day or night
;                              in its _pbr.txt, or a rule of field_time.txt) they are fought by day.
; time        = cycle | day | night   cycle: the day/night clock; day or night: always that
; day_tint    = what the picture is multiplied by in daylight
; night_tint, night_level = by night the picture is multiplied by night_level x night_tint
; ao          = how much of a page's ORM map's AO is used (0-1)
; glow        = how bright a page's glows are at night (<page>_EC, a world page's _E; 1: as painted)
; A page's maps are next to its picture: a battle stage page's <page>_N (normals, green up), _ORMH (AO, roughness,
; metalness, and the height in its alpha: 1 the surface; or _ORM without it) and _EC (a picture of its glows, with
; their coverage); a world map page's (texl_003_0, wmset_016_0 ...) _N, _ORMH and _E (the glows' colour on black,
; added at night) or _EC. A palette's own world maps (texl_003_2_N), else the first palette's (texl_003_0_N).

[battle]
night = auto
time = cycle
day_tint = 1 1 1
night_tint = 0.55 0.65 1
night_level = 0.3
ao = 0.5
glow = 1
; The sun and the moon cross the stage by the clock (the camera's place in the stage is read from the game):
; sun         = how much a surface turned away from it darkens (0-1; 0: the stage as painted)
; relief      = how strongly a page's normal map (<page>_N) shows in its light (0-4)
; specular    = its highlight on pages with an ORM map (0-4)
; parallax    = how deep the height of an ORMH map goes, in the place's units (0-64; 0: off): lower parts shift
;               with the view
; sun_azimuth = where it rises, degrees around the stage (0: along the stage's +x); sun_height = how high it climbs
; fighter_sun = how much of that darkening the fighters get (0-1; their normals are smoothed; 0: as the game lights them)
; shadow      = how much of the sun the fighters' shadows take off the stage (0-1; 0: no shadows)
sun = 0.35
fighter_sun = 0.5
shadow = 0.5
relief = 1
specular = 0.5
parallax = 4
sun_azimuth = 30
sun_height = 60

[world]
night = auto
time = cycle
day_tint = 1 1 1
night_tint = 0.55 0.65 1
night_level = 0.3
; The sun and the moon cross the world map by the clock too (its camera is read from the game): sun, sun_azimuth
; and sun_height as for battles. The ground is big flat triangles, each lit flat, so it darkens less here
sun = 0.25
sun_azimuth = 30
sun_height = 60
; its pages' maps, as for battles
ao = 0.5
glow = 1
relief = 1
specular = 0.5
parallax = 4

; Rules change the settings above where their conditions hold. Only the keys written in a rule change anything.
;   where  = battle | world | both     (left out: both)
;   stage  = 12, 40-45                 battle stages by number (a0stg012 is 12); left out: every stage. A rule
;                                      with stages is for battles only
;   moment = 3000-65535                the story's progress, the game's own counter (the log says what it is when a
;                                      battle or the world map starts, with SCENE_DEBUG=1); left out: always
;   light  = x up z  r g b  range [night|day]
;            a light of the stage's own, four per stage at most. x and z: its place on the stage's floor, in the
;            stage's units (0 0 is the middle; the fighters stand a few thousand apart); up: how far above the
;            floor; r g b: its colour (linear, may be above 1); range: how far it reaches; night or day: only then
; Later rules go over earlier ones.
