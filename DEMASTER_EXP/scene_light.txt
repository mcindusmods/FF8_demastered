; DEMASTER: the time of day and colour overrides in battles and on the world map (SCENE_LIGHT in demaster.conf).
; Read again within a second of saving it. Colours are red green blue in linear light: 1 1 1 changes nothing.
;
; night       = auto | 0 | 1   does the night come here. auto: the world map always; a battle when it started on
;                              the world map, in a field open to the sky (sky = 1 in its _pbr.txt) or in a field
;                              whose _pbr.txt says time = night. Battles from other fields stay as painted.
; time        = cycle | day | night   cycle: the day/night clock; day or night: always that
; day_tint    = what the picture is multiplied by in daylight
; night_tint, night_level = by night the picture is multiplied by night_level x night_tint
; ao          = how much of a stage page's ORM map's AO is used (0-1)
; glow        = how bright a stage page's picture of its glows is (<page>_EC; 1: as painted)

[battle]
night = auto
time = cycle
day_tint = 1 1 1
night_tint = 0.55 0.65 1
night_level = 0.3
ao = 0.5
glow = 1
; The sun and the moon cross the stage by the clock (the camera's place in the stage is read from the game):
; sun         = how much a surface turned away from it darkens (0-1; 0: the stage as painted)
; relief      = how strongly a stage page's normal map (<page>_N) shows in its light (0-4)
; specular    = its highlight on pages with an ORM map (0-4)
; sun_azimuth = where it rises, degrees around the stage (0: along the stage's +x); sun_height = how high it climbs
sun = 0.35
relief = 1
specular = 0.5
sun_azimuth = 30
sun_height = 60

[world]
night = auto
time = cycle
day_tint = 1 1 1
night_tint = 0.55 0.65 1
night_level = 0.3

; Rules change the settings above where their conditions hold. Only the keys written in a rule change anything.
;   where  = battle | world | both     (left out: both)
;   stage  = 12, 40-45                 battle stages by number (a0stg012 is 12); left out: every stage. A rule
;                                      with stages is for battles only
;   moment = 3000-65535                the story's progress, the game's own counter (the log says what it is when a
;                                      battle or the world map starts, with SCENE_DEBUG=1); left out: always
;   light  = x up z  r g b  range [night|day]
;            a light of the stage's own, four per stage at most. x and z: its place on the stage's floor, in the
;            stage's units (0 0 is the middle; the fighters stand a few thousand apart); up: how far above the
;            floor; r g b: its colour (linear, may be above 1); range: how far it reaches; night or day: only then
; Later rules go over earlier ones.

; After the Lunar Cry (the counter is 3000 once the Ragnarok is yours): the night is red instead of blue.
; Change the colours to taste, or delete the rule.
[rule lunar cry]
where = both
moment = 3000-65535
day_tint = 1 0.94 0.92
night_tint = 1 0.5 0.45
